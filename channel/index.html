>as an example, a program compiled with all of the implementation and then placed under /bin/cat, would read it’s name then run the cat function inside the program.

here’s the snippet of the code:

type StartFn = fn() -> Result<i32, Box<dyn core::error::Error>>;
const IMPL_TABLES: &[(&str, StartFn)] = &[
    ("cat", coreutils::cat::start),
    ...
];

#[unsafe(no_mangle)]
pub extern "C" fn main() -> i32 {
    let mut args = env::args();
    let program_name =
        args.next().unwrap_or_default().split("/").last().unwrap_or_default();

    for (name, start_fn) in IMPL_TABLES {
        if *name == program_name {
            match start_fn() {
                Ok(exit_code) => process::exit(exit_code as u64),
                Err(e) => {
                    println!("Error: {}", e);
                    process::exit(1);
                },
            }
        }
    }
    0
}
